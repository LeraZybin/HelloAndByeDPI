cmake_minimum_required(VERSION 3.20)
project(ByeDPI_Desktop VERSION 0.6.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "HelloAndByeDPI supports only 64-bit Windows")
  endif()
  enable_language(RC)
  add_compile_definitions(_WIN32_WINNT=0x0602 WINVER=0x0602)
endif()
enable_testing()
add_executable(argument_tests tests/arguments.cpp)
target_include_directories(argument_tests PRIVATE src)
add_test(NAME arguments COMMAND argument_tests)
add_executable(proxy_transaction_tests tests/proxy_transaction.cpp)
target_include_directories(proxy_transaction_tests PRIVATE src)
add_test(NAME proxy_transaction COMMAND proxy_transaction_tests)

set(BYE_SOURCE "${CMAKE_SOURCE_DIR}/third_party/byedpi")
if(NOT EXISTS "${BYE_SOURCE}/main.c")
  message(FATAL_ERROR "Run python3 scripts/fetch_dependencies.py first")
endif()
add_executable(ciadpi ${BYE_SOURCE}/main.c ${BYE_SOURCE}/conev.c
  ${BYE_SOURCE}/proxy.c ${BYE_SOURCE}/desync.c ${BYE_SOURCE}/packets.c
  ${BYE_SOURCE}/mpool.c ${BYE_SOURCE}/extend.c)
target_compile_definitions(ciadpi PRIVATE _DEFAULT_SOURCE)
if(WIN32)
  target_sources(ciadpi PRIVATE ${BYE_SOURCE}/win_service.c)
  target_link_libraries(ciadpi PRIVATE ws2_32 mswsock)
  if(MINGW)
    target_link_options(ciadpi PRIVATE -static)
  endif()
endif()

set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
set(wxUSE_LIBTIFF OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/wxWidgets EXCLUDE_FROM_ALL)
add_executable(ui_string_tests tests/ui_strings.cpp)
target_include_directories(ui_string_tests PRIVATE src)
target_link_libraries(ui_string_tests PRIVATE wx::core wx::base)
add_test(NAME ui_strings COMMAND ui_string_tests)
if(APPLE AND TARGET wxpng)
  # New Apple SDKs define TARGET_OS_MAC in C headers. The bundled libpng
  # mistakes this for classic Mac OS; preinclude the modern math header.
  target_compile_options(wxpng PRIVATE -include math.h)
endif()
add_library(http_probe STATIC src/http_probe.cpp)
target_include_directories(http_probe PUBLIC src)
if(APPLE)
  # Use the OS libcurl, including the system trust store; APIs used exist on 10.13.
  target_link_libraries(http_probe PUBLIC "-lcurl")
elseif(WIN32)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
  set(HTTP_ONLY ON CACHE BOOL "" FORCE)
  set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
  set(ENABLE_THREADED_RESOLVER ON CACHE BOOL "" FORCE)
  foreach(option CURL_ZLIB CURL_BROTLI CURL_ZSTD CURL_USE_LIBPSL CURL_USE_LIBSSH2 USE_NGHTTP2 USE_LIBIDN2)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory(third_party/curl EXCLUDE_FROM_ALL)
  target_link_libraries(http_probe PUBLIC CURL::libcurl)
  target_link_libraries(http_probe PUBLIC dnsapi ws2_32)
endif()
add_executable(http_probe_tests tests/http_probe.cpp)
target_link_libraries(http_probe_tests PRIVATE http_probe)
add_test(NAME http_probe_model COMMAND http_probe_tests "${CMAKE_SOURCE_DIR}/data/byebyedpi/proxytest_strategies.list")
add_executable(probe_integration tests/probe_integration.cpp)
target_link_libraries(probe_integration PRIVATE http_probe)
add_executable(ByeDPI WIN32 MACOSX_BUNDLE src/main.cpp src/proxy_journal.cpp src/policy_test.cpp)
if(WIN32)
  configure_file(src/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  target_sources(ByeDPI PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc src/system_proxy_win.cpp src/vpn_win.cpp)
  target_link_libraries(ByeDPI PRIVATE wininet iphlpapi shell32)
  add_executable(ByeDPILauncher WIN32 src/launcher_win.c ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  set_target_properties(ByeDPILauncher PROPERTIES OUTPUT_NAME "HelloAndByeDPI-launcher")
  set(BYEDPI_WX_SETUP_DIR "${wxWidgets_BINARY_DIR}/lib/gcc_x64_lib/mswu")
  target_include_directories(ByeDPILauncher PRIVATE
    "${CMAKE_SOURCE_DIR}/third_party/wxWidgets/include"
    "${BYEDPI_WX_SETUP_DIR}")
  if(MINGW)
    target_link_options(ByeDPILauncher PRIVATE -nostdlib -Wl,-e,wWinMainCRTStartup -Wl,--subsystem,windows)
    target_link_libraries(ByeDPILauncher PRIVATE kernel32 shell32 user32)
  endif()
  add_executable(windows_proxy_tests tests/windows_proxy.cpp src/system_proxy_win.cpp)
  target_include_directories(windows_proxy_tests PRIVATE src)
  target_link_libraries(windows_proxy_tests PRIVATE wx::base wininet)
  add_test(NAME windows_proxy_roundtrip COMMAND windows_proxy_tests)
elseif(APPLE)
  target_sources(ByeDPI PRIVATE src/system_proxy_mac.cpp src/vpn_mac.cpp assets/icon.icns)
  set_source_files_properties(assets/icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
  target_link_libraries(ByeDPI PRIVATE "-framework SystemConfiguration" "-framework Security")
else()
  message(FATAL_ERROR "System proxy integration currently supports Windows and macOS")
endif()
target_link_libraries(ByeDPI PRIVATE wx::core wx::base wx::net http_probe)
add_executable(proxy_journal_tests tests/proxy_journal.cpp src/proxy_journal.cpp)
target_include_directories(proxy_journal_tests PRIVATE src)
target_link_libraries(proxy_journal_tests PRIVATE wx::base)
add_test(NAME proxy_journal COMMAND proxy_journal_tests)
if(APPLE)
  add_executable(mac_proxy_tests src/system_proxy_mac.cpp)
  target_compile_definitions(mac_proxy_tests PRIVATE BYEDPI_PROXY_TEST)
  target_link_libraries(mac_proxy_tests PRIVATE wx::base "-framework SystemConfiguration" "-framework Security")
  add_test(NAME mac_proxy_values COMMAND mac_proxy_tests)
  add_executable(mac_vpn_tests tests/vpn_mac.cpp src/vpn_mac.cpp)
  target_include_directories(mac_vpn_tests PRIVATE src)
  target_link_libraries(mac_vpn_tests PRIVATE wx::base "-framework SystemConfiguration")
  add_test(NAME mac_vpn_physical_network COMMAND mac_vpn_tests)
endif()
if(MINGW)
  target_link_options(ByeDPI PRIVATE -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(ByeDPI PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.byedpi.desktop"
  OUTPUT_NAME "HelloAndByeDPI"
  MACOSX_BUNDLE_ICON_FILE "icon.icns"
  MACOSX_BUNDLE_BUNDLE_NAME "HelloAndByeDPI"
  MACOSX_BUNDLE_BUNDLE_VERSION "0.6.3"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "0.6.3")
add_dependencies(ByeDPI ciadpi)
add_custom_command(TARGET ByeDPI POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:ciadpi> $<TARGET_FILE_DIR:ByeDPI>
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ByeDPI>/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:ByeDPI>/data"
  COMMAND ${CMAKE_COMMAND} -E copy "${BYE_SOURCE}/LICENSE" "$<TARGET_FILE_DIR:ByeDPI>/licenses/ByeDPI.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/wxWidgets/docs/licence.txt" "$<TARGET_FILE_DIR:ByeDPI>/licenses/wxWidgets.txt")
foreach(license docs/lgpl.txt src/png/LICENSE src/jpeg/README src/zlib/LICENSE
    src/expat/expat/COPYING 3rdparty/pcre/LICENCE)
  string(REPLACE "/" "-" filename "${license}")
  add_custom_command(TARGET ByeDPI POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_SOURCE_DIR}/third_party/wxWidgets/${license}"
    "$<TARGET_FILE_DIR:ByeDPI>/licenses/${filename}")
endforeach()
if(WIN32)
  add_custom_command(TARGET ByeDPI POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_SOURCE_DIR}/third_party/curl/COPYING" "$<TARGET_FILE_DIR:ByeDPI>/licenses/curl.txt")
  add_custom_command(TARGET ByeDPI POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/vpn/windows-x64/tun2socks.exe" "$<TARGET_FILE_DIR:ByeDPI>/tun2socks.exe"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/vpn/windows-x64/wintun.dll" "$<TARGET_FILE_DIR:ByeDPI>/wintun.dll"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/vpn/licenses/tun2socks.txt" "$<TARGET_FILE_DIR:ByeDPI>/licenses/tun2socks.txt"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/vpn/licenses/Wintun.txt" "$<TARGET_FILE_DIR:ByeDPI>/licenses/Wintun.txt")
elseif(APPLE)
  if(CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    set(BYEDPI_MAC_VPN_ARCH arm64)
  elseif(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
    set(BYEDPI_MAC_VPN_ARCH x86_64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(BYEDPI_MAC_VPN_ARCH arm64)
  else()
    set(BYEDPI_MAC_VPN_ARCH x86_64)
  endif()
  set(BYEDPI_MAC_TUN2SOCKS "${CMAKE_SOURCE_DIR}/third_party/vpn/macos-${BYEDPI_MAC_VPN_ARCH}/tun2socks")
  if(NOT EXISTS "${BYEDPI_MAC_TUN2SOCKS}")
    message(FATAL_ERROR "Run python3 scripts/fetch_macos_vpn.py ${BYEDPI_MAC_VPN_ARCH} first")
  endif()
  add_custom_command(TARGET ByeDPI POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ByeDPI>/lib"
    COMMAND ${CMAKE_COMMAND} -E copy "${BYEDPI_MAC_TUN2SOCKS}" "$<TARGET_FILE_DIR:ByeDPI>/lib/tun2socks"
    COMMAND chmod 755 "$<TARGET_FILE_DIR:ByeDPI>/lib/tun2socks"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/vpn/licenses/tun2socks.txt" "$<TARGET_FILE_DIR:ByeDPI>/licenses/tun2socks.txt")
endif()
